# 前端界面与日志管理

## 🎨 Web控制台概述

GeminiCLI to API 提供了一个功能完整的Web控制台，使用户能够轻松管理OAuth认证、凭证文件、使用统计和系统配置。控制台采用响应式设计，支持桌面和移动设备，提供直观的用户界面和实时数据更新。

### 控制台架构

```
用户界面 → 前端交互 → API请求 → 后端处理 → 数据更新
    ↓         ↓         ↓         ↓         ↓
页面渲染 ← 状态更新 ← 响应数据 ← 业务逻辑 ← 数据存储
```

### 前端技术栈

| 技术 | 用途 | 特点 |
|------|------|------|
| **HTML5** | 页面结构 | 语义化标签、响应式布局 |
| **CSS3** | 样式设计 | Flexbox布局、动画效果、移动端适配 |
| **JavaScript** | 交互逻辑 | 异步请求、DOM操作、事件处理 |
| **WebSocket** | 实时通信 | 日志流、状态更新 |
| **Font Awesome** | 图标库 | 丰富的图标资源 |

```javascript
// 开始认证流程
async function startAuth() {
    try {
        showLoading('getAuthBtn', '正在获取认证链接...');
        
        const response = await fetch('/auth/start', {
            method: 'POST',
            headers: {
                'Authorization': `Bearer ${authToken}`,
                'Content-Type': 'application/json'
            },
            body: JSON.stringify({
                project_id: null,  // 自动检测
                get_all_projects: true  // 获取所有项目
            })
        });
        
        const data = await response.json();
        
        if (response.ok) {
            // 显示认证链接
            document.getElementById('authUrl').href = data.auth_url;
            document.getElementById('authUrlSection').classList.remove('hidden');
            
            // 存储状态
            currentAuthState = data.state;
            
            showNotification('success', '认证链接已生成，请点击链接进行授权');
        } else {
            showNotification('error', data.error || '获取认证链接失败');
        }
    } catch (error) {
        showNotification('error', '请求失败: ' + error.message);
    } finally {
        hideLoading('getAuthBtn', '<i class="fas fa-link"></i> 获取认证链接');
    }
}

// 处理回调URL
async function processCallbackUrl() {
    try {
        const callbackUrl = document.getElementById('callbackUrlInput').value.trim();
        
        if (!callbackUrl) {
            showNotification('error', '请输入回调URL');
            return;
        }
        
        showLoading('submitCallbackBtn', '正在处理认证...');
        
        const response = await fetch('/auth/callback-url', {
            method: 'POST',
            headers: {
                'Authorization': `Bearer ${authToken}`,
                'Content-Type': 'application/json'
            },
            body: JSON.stringify({
                callback_url: callbackUrl,
                get_all_projects: true
            })
        });
        
        const data = await response.json();
        
        if (response.ok) {
            if (data.multiple_credentials) {
                // 批量认证成功
                handleMultipleCredentials(data.multiple_credentials);
            } else {
                // 单项目认证成功
                handleSingleCredential(data);
            }
            
            showNotification('success', '认证成功，凭证已保存');
        } else {
            if (data.requires_project_selection) {
                // 需要选择项目
                showProjectSelection(data.available_projects);
            } else if (data.requires_manual_project_id) {
                // 需要手动输入项目ID
                showManualProjectIdInput();
            } else {
                showNotification('error', data.error || '认证失败');
            }
        }
    } catch (error) {
        showNotification('error', '请求失败: ' + error.message);
    } finally {
        hideLoading('submitCallbackBtn', '<i class="fas fa-paper-plane"></i> 提交URL');
    }
}

// 处理多个凭证
function handleMultipleCredentials(multipleCredentials) {
    const successCount = multipleCredentials.success.length;
    const failedCount = multipleCredentials.failed.length;
    
    let message = `批量认证完成：成功 ${successCount} 个，失败 ${failedCount} 个`;
    
    if (successCount > 0) {
        message += '\n\n成功认证的项目：';
        multipleCredentials.success.forEach(cred => {
            message += `\n- ${cred.project_name} (${cred.project_id})`;
        });
    }
    
    if (failedCount > 0) {
        message += '\n\n失败的项目：';
        multipleCredentials.failed.forEach(cred => {
            message += `\n- ${cred.project_name} (${cred.project_id}): ${cred.error}`;
        });
    }
    
    showNotification('info', message);
    
    // 刷新凭证列表
    if (currentTab === 'manage') {
        refreshCredsStatus();
    }
}

// 处理单个凭证
function handleSingleCredential(data) {
    // 显示凭证内容
    const credentialsContent = document.getElementById('credentialsContent');
    credentialsContent.textContent = JSON.stringify(data.credentials, null, 2);
    document.getElementById('credentialsSection').classList.remove('hidden');
    
    // 刷新凭证列表
    if (currentTab === 'manage') {
        refreshCredsStatus();
    }
}
```

```javascript
// 刷新凭证状态
async function refreshCredsStatus() {
    try {
        showLoading('credsLoading');
        
        const response = await fetch('/creds/status', {
            headers: {
                'Authorization': `Bearer ${authToken}`
            }
        });
        
        const data = await response.json();
        
        if (response.ok) {
            displayCredsList(data.creds);
            updateStats(data.creds);
        } else {
            showNotification('error', data.error || '获取凭证状态失败');
        }
    } catch (error) {
        showNotification('error', '请求失败: ' + error.message);
    } finally {
        hideLoading('credsLoading');
    }
}

// 显示凭证列表
function displayCredsList(creds) {
    const credsList = document.getElementById('credsList');
    credsList.innerHTML = '';
    
    // 应用筛选
    const filteredCreds = filterCreds(creds);
    
    // 应用分页
    const pageSize = parseInt(document.getElementById('pageSizeSelect').value);
    const currentPage = parseInt(window.sessionStorage.getItem('credsCurrentPage') || '1');
    const startIndex = (currentPage - 1) * pageSize;
    const endIndex = startIndex + pageSize;
    const pageCreds = filteredCreds.slice(startIndex, endIndex);
    
    // 创建凭证项
    pageCreds.forEach(([filename, credInfo]) => {
        const credItem = createCredItem(filename, credInfo);
        credsList.appendChild(credItem);
    });
    
    // 更新分页信息
    updatePagination(filteredCreds.length, currentPage, pageSize);
}

// 创建凭证项

// 切换凭证状态
async function toggleCredStatus(filename, enable) {
    try {
        const action = enable ? 'enable' : 'disable';
        
        const response = await fetch('/creds/action', {
            method: 'POST',
            headers: {
                'Authorization': `Bearer ${authToken}`,
                'Content-Type': 'application/json'
            },
            body: JSON.stringify({
                filename: filename,
                action: action
            })
        });
        
        const data = await response.json();
        
        if (response.ok) {
            showNotification('success', data.message);
            refreshCredsStatus();
        } else {
            showNotification('error', data.error || '操作失败');
        }
    } catch (error) {
        showNotification('error', '请求失败: ' + error.message);
    }
}

// 批量操作
async function batchAction(action) {
    try {
        const selectedCheckboxes = document.querySelectorAll('.cred-select-checkbox:checked');
        const filenames = Array.from(selectedCheckboxes).map(cb => cb.value);
        
        if (filenames.length === 0) {
            showNotification('warning', '请选择要操作的文件');
            return;
        }
        
        const actionText = {
            'enable': '启用',
            'disable': '禁用',
            'delete': '删除'
        }[action];
        
        if (action === 'delete' && !confirm(`确定要删除选中的 ${filenames.length} 个文件吗？`)) {
            return;
        }
        
        const response = await fetch('/creds/batch-action', {
            method: 'POST',
            headers: {
                'Authorization': `Bearer ${authToken}`,
                'Content-Type': 'application/json'
            },
            body: JSON.stringify({
                action: action,
                filenames: filenames
            })
        });
        
        const data = await response.json();
        
        if (response.ok) {
            showNotification('success', data.message);
            refreshCredsStatus();
        } else {
            showNotification('error', data.error || '批量操作失败');
        }
    } catch (error) {
        showNotification('error', '请求失败: ' + error.message);
    }
}

// 下载所有凭证
async function downloadAllCreds() {
    try {
        const response = await fetch('/creds/download-all', {
            headers: {
                'Authorization': `Bearer ${authToken}`
            }
        });
        
        if (response.ok) {
            const blob = await response.blob();
            const url = window.URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = 'credentials.zip';
            document.body.appendChild(a);
            a.click();
            window.URL.revokeObjectURL(url);
            document.body.removeChild(a);
            
            showNotification('success', '凭证文件已下载');
        } else {
            const data = await response.json();
            showNotification('error', data.error || '下载失败');
        }
    } catch (error) {
        showNotification('error', '下载失败: ' + error.message);
    }
}
```

```python
class Logger:
    """支持 log('info', 'msg') 和 log.info('msg') 两种调用方式"""
    
    def __call__(self, level: str, message: str):
        """支持 log('info', 'message') 调用方式"""
        _log(level, message)

    def debug(self, message: str):
        """记录调试信息"""
        _log('debug', message)
    
    def info(self, message: str):
        """记录一般信息"""
        _log('info', message)
    
    def warning(self, message: str):
        """记录警告信息"""
        _log('warning', message)
    
    def error(self, message: str):
        """记录错误信息"""
        _log('error', message)
    
    def critical(self, message: str):
        """记录严重错误信息"""
        _log('critical', message)
    
    def get_current_level(self) -> str:
        """获取当前日志级别名称"""
        current_level = _get_current_log_level()
        for name, value in LOG_LEVELS.items():
            if value == current_level:
                return name
        return 'info'
    
    def get_log_file(self) -> str:
        """获取当前日志文件路径"""
        return _get_log_file_path()
```

### 日志写入机制

```python
def _write_to_file(message: str):
    """线程安全地写入日志文件"""
    global _file_writing_disabled, _disable_reason
    
    # 如果文件写入已被禁用，直接返回
    if _file_writing_disabled:
        return
    
    try:
        log_file = _get_log_file_path()
        with _file_lock:
            with open(log_file, 'a', encoding='utf-8') as f:
                f.write(message + '\n')
                f.flush()  # 强制刷新到磁盘，确保实时写入
    except (PermissionError, OSError, IOError) as e:
        # 检测只读文件系统或权限问题，禁用文件写入
        _file_writing_disabled = True
        _disable_reason = str(e)
        print(f"Warning: File system appears to be read-only or permission denied. Disabling log file writing: {e}", file=sys.stderr)
        print(f"Log messages will continue to display in console only.", file=sys.stderr)
    except Exception as e:
        # 其他异常仍然输出警告但不禁用写入（可能是临时问题）
        print(f"Warning: Failed to write to log file: {e}", file=sys.stderr)
```

### WebSocket日志端点

**文件位置**：`src/web_routes.py`

```python
@router.websocket("/auth/logs/stream")
async def websocket_logs(websocket: WebSocket):
    """WebSocket端点，用于实时日志流"""
    # 检查连接数限制
    if not await manager.connect(websocket):
        return
    
    try:
        # 直接使用环境变量获取日志文件路径
        log_file_path = os.getenv('LOG_FILE', 'log.txt')
        
        # 发送初始日志（限制为最后50行，减少内存占用）
        if os.path.exists(log_file_path):
            try:
                with open(log_file_path, "r", encoding="utf-8") as f:
                    lines = f.readlines()
                    # 只发送最后50行，减少初始内存消耗
                    for line in lines[-50:]:
                        if line.strip():
                            await websocket.send_text(line.strip())
            except Exception as e:
                await websocket.send_text(f"Error reading log file: {e}")
        
        # 监控日志文件变化
        last_size = os.path.getsize(log_file_path) if os.path.exists(log_file_path) else 0
        max_read_size = 8192  # 限制单次读取大小为8KB，防止大量日志造成内存激增
        check_interval = 2    # 增加检查间隔，减少CPU和I/O开销
        
        while websocket.client_state == WebSocketState.CONNECTED:
            await asyncio.sleep(check_interval)
            
            if os.path.exists(log_file_path):
                current_size = os.path.getsize(log_file_path)
                if current_size > last_size:
                    # 限制读取大小，防止单次读取过多内容
                    read_size = min(current_size - last_size, max_read_size)
                    
                    try:
                        with open(log_file_path, "r", encoding="utf-8", errors="replace") as f:
                            f.seek(last_size)
                            new_content = f.read(read_size)
                            
                            # 处理编码错误的情况
                            if not new_content:
                                last_size = current_size
                                continue
                            
                            # 分行发送，避免发送不完整的行
                            lines = new_content.splitlines(keepends=True)
                            if lines:
                                # 如果最后一行没有换行符，保留到下次处理
                                if not lines[-1].endswith('\n') and len(lines) > 1:
                                    # 除了最后一行，其他都发送
                                    for line in lines[:-1]:
                                        if line.strip():
                                            await websocket.send_text(line.rstrip())
                                    # 更新位置，但要退回最后一行的字节数
                                    last_size += len(new_content.encode('utf-8')) - len(lines[-1].encode('utf-8'))
                                else:
                                    # 所有行都发送
                                    for line in lines:
                                        if line.strip():
                                            await websocket.send_text(line.rstrip())
                                    last_size += len(new_content.encode('utf-8'))
                    except UnicodeDecodeError as e:
                        # 遇到编码错误时，跳过这部分内容
                        log.warning(f"WebSocket日志读取编码错误: {e}, 跳过部分内容")
                        last_size = current_size
                    except Exception as e:
                        await websocket.send_text(f"Error reading new content: {e}")
                        # 发生其他错误时，重置文件位置
                        last_size = current_size
                        
                # 如果文件被截断（如清空日志），重置位置
                elif current_size < last_size:
                    last_size = 0
                    await websocket.send_text("--- 日志已清空 ---")
                    
    except WebSocketDisconnect:
        pass
    except Exception as e:
        log.error(f"WebSocket logs error: {e}")
    finally:
        manager.disconnect(websocket)
```

```javascript
// WebSocket日志流
let logWebSocket = null;
let logReconnectTimer = null;
let logReconnectAttempts = 0;
const MAX_LOG_RECONNECT_ATTEMPTS = 5;

// 连接日志WebSocket
function connectLogWebSocket() {
    const protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
    const wsUrl = `${protocol}//${window.location.host}/auth/logs/stream`;
    
    try {
        logWebSocket = new WebSocket(wsUrl);
        
        logWebSocket.onopen = function(event) {
            console.log('日志WebSocket连接已建立');
            logReconnectAttempts = 0;
            
            // 清除重连定时器
            if (logReconnectTimer) {
                clearTimeout(logReconnectTimer);
                logReconnectTimer = null;
            }
        };
        
        logWebSocket.onmessage = function(event) {
            const logMessage = event.data;
            appendLogMessage(logMessage);
        };
        
        logWebSocket.onclose = function(event) {
            console.log('日志WebSocket连接已关闭');
            
            // 尝试重连
            if (logReconnectAttempts < MAX_LOG_RECONNECT_ATTEMPTS) {
                logReconnectAttempts++;
                const delay = Math.min(1000 * Math.pow(2, logReconnectAttempts), 30000);
                
                console.log(`日志WebSocket重连尝试 ${logReconnectAttempts}/${MAX_LOG_RECONNECT_ATTEMPTS}，${delay}ms后重试`);
                
                logReconnectTimer = setTimeout(connectLogWebSocket, delay);
            } else {
                console.error('日志WebSocket重连次数已达上限，停止重连');
                appendLogMessage('--- 日志WebSocket连接已断开，请刷新页面重新连接 ---', 'error');
            }
        };
        
        logWebSocket.onerror = function(error) {
            console.error('日志WebSocket错误:', error);
        };
        
    } catch (error) {
        console.error('创建日志WebSocket连接失败:', error);
    }
}

// 断开日志WebSocket
function disconnectLogWebSocket() {
    if (logWebSocket) {
        logWebSocket.close();
        logWebSocket = null;
    }
    
    if (logReconnectTimer) {
        clearTimeout(logReconnectTimer);
        logReconnectTimer = null;
    }
}

// 添加日志消息
function appendLogMessage(message, type = 'info') {
    // 如果当前不在日志标签页，不处理
    if (currentTab !== 'logs') {
        return;
    }
    
    const logContainer = document.getElementById('logContainer');
    if (!logContainer) {
        return;
    }
    
    // 创建日志行
    const logLine = document.createElement('div');
    logLine.className = `log-line log-${type}`;
    
    // 添加时间戳
    const timestamp = new Date().toLocaleTimeString();
    logLine.innerHTML = `<span class="log-timestamp">[${timestamp}]</span> ${escapeHtml(message)}`;
    
    // 添加到容器
    logContainer.appendChild(logLine);
    
    // 自动滚动到底部
    logContainer.scrollTop = logContainer.scrollHeight;
    
    // 限制日志行数，防止内存溢出
    const maxLogLines = 1000;
    const logLines = logContainer.children;
    if (logLines.length > maxLogLines) {
        logContainer.removeChild(logLines[0]);
    }
}

async function downloadLogs() {
    try {
        const response = await fetch('/auth/logs/download', {
            headers: {
                'Authorization': `Bearer ${authToken}`
            }
        });
        
        if (response.ok) {
            const blob = await response.blob();
            const url = window.URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = response.headers.get('Content-Disposition').match(/filename=(.+)/)[1];
            document.body.appendChild(a);
            a.click();
            window.URL.revokeObjectURL(url);
            document.body.removeChild(a);
            
            showNotification('success', '日志文件已下载');
        } else {
            const data = await response.json();
            showNotification('error', data.error || '下载日志失败');
        }
    } catch (error) {
        showNotification('error', '下载失败: ' + error.message);
    }
}

// HTML转义
function escapeHtml(text) {
    const map = {
        '&': '&',
        '<': '<',
        '>': '>',
        '"': '"',
        "'": '&#039;'
    };
    return text.replace(/[&<>"']/g, m => map[m]);
}
```

## 📱 移动端适配

### 响应式设计原则

1. **流式布局**：使用Flexbox和Grid布局，自动适应不同屏幕尺寸
2. **弹性单位**：使用百分比、em、rem等相对单位
3. **媒体查询**：针对不同屏幕尺寸应用不同样式
4. **触摸优化**：增大按钮和交互元素的点击区域
5. **简化界面**：在小屏幕上隐藏或简化非必要元素

### 移动端特殊处理

```javascript
// 检测移动设备
function isMobileDevice() {
    return /Android|webOS|iPhone|iPad|iPod|BlackBerry|IEMobile|Opera Mini/i.test(navigator.userAgent);
}

// 移动端适配
if (isMobileDevice()) {
    // 添加移动端类
    document.body.classList.add('mobile-device');
    
    // 调整触摸事件
    document.addEventListener('touchstart', function() {}, {passive: true});
    
    // 禁用双击缩放
    let lastTouchEnd = 0;
    document.addEventListener('touchend', function(event) {
        const now = Date.now();
        if (now - lastTouchEnd <= 300) {
            event.preventDefault();
        }
        lastTouchEnd = now;
    }, false);
}
```
